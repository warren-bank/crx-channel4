const print_api_1 = (id) => {
  fetch(`https://www.channel4.com/vod/stream/${id}`)
    .then(res => res.json())
    .then(data => data.videoProfiles.map(profile => profile.streams.map(stream => ({uri: stream.uri, token: stream.token}))).reduce((all, some) => [...all, ...some], []))
    .then(streams => {console.log(JSON.stringify(streams, null, 2))})
}

const print_api_2 = (id) => {
  fetch(`https://api.channel4.com/online/v1/vod/stream/${id}?client=android-mod`)
    .then(res => res.text())
    .then(console.log)
}
